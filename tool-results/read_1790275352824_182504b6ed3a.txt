   159→function computeUrgency(days: number): Urgency {
   160→  if (days < 0) return "past";
   161→  if (days === 0) return "today";
   162→  if (days <= 1) return "critical";
   163→  if (days <= 3) return "urgent";
   164→  if (days <= 7) return "soon";
   165→  return "normal";
   166→}
   167→
   168→/** Normalize a raw briefing (from listBriefings) into a BriefingAlert. */
   169→function toBriefingView(b: any): BriefingAlert {
   170→  const days = b.days_until ?? daysUntil(b.date) ?? -999;
   171→  return {
   172→    id: b.id,
   173→    tender_id: b.tender_id,
   174→    tender_reference: b.tender_reference ?? null,
   175→    tender_title: b.tender_title ?? null,
   176→    client: b.client ?? null,
   177→    briefing_type: b.briefing_type as BriefingType,
   178→    date: b.date ?? null,
   179→    time: b.time ?? null,
   180→    location: b.location ?? null,
   181→    contact_person: b.contact_person ?? null,
   182→    attendance_required: !!b.attendance_required,
   183→    attendance_confirmed: !!b.attendance_confirmed,
   184→    person_attending: b.person_attending ?? null,
   185→    certificate_path: b.certificate_path ?? null,
   186→    certificate_url: b.certificate_url ?? null,
   187→    notes: b.notes ?? null,
   188→    days_until: days,
   189→    is_past: days < 0,
   190→    is_today: days === 0,
   191→    urgency: (b.urgency as Urgency) ?? computeUrgency(days),
   192→  };
   193→}

   387→  const config = URGENCY_CONFIG[b.urgency];
   388→  const UrgencyIcon = config.icon;
   389→  const isPast = b.is_past;
   390→  const showStrip = b.urgency === "today" || b.urgency === "critical" || b.urgency === "urgent" || b.urgency === "soon";
   391→  const needsConfirmation =
   392→    b.attendance_required &&
   393→    !b.attendance_confirmed &&
   394→    !isPast &&
   395→    b.days_until >= 0 &&
   396→    b.days_until <= 7;
   397→  const noAttendee = b.attendance_required && !b.person_attending && !isPast;

   400→
   401→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   402→    const f = e.target.files?.[0];
   403→    if (f) onUploadCertificate(b, f);

   405→  };
   406→
   407→  const countdownLabel =
   408→    b.days_until === 0
   409→      ? "Today"
   410→      : b.days_until === 1
   411→      ? "1 day left"

   413→      ? `${b.days_until} days left`

   417→    <Card
   418→      className={cn(
   419→        "flex flex-col transition-shadow hover:shadow-md overflow-hidden",
