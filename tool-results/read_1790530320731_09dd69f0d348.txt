    20→def _run_zai_chat(prompt: str, system: Optional[str] = None, max_tokens: int = 8000) -> str:
    21→    """Run the z-ai chat CLI and return the assistant text content."""
    22→    with tempfile.NamedTemporaryFile(mode="w", suffix=".json", delete=False) as f:
    23→        out_path = f.name
    24→    try:
    25→        cmd = [Z_AI_CLI, "chat", "--prompt", prompt, "--output", out_path]
    26→        if system:
    27→            cmd += ["--system", system]
    28→        proc = subprocess.run(cmd, capture_output=True, text=True, timeout=180)
    29→        if proc.returncode != 0:
    30→            raise RuntimeError(f"z-ai chat failed: {proc.stderr[:500]}")
    31→        with open(out_path, "r") as fp:
    32→            data = json.load(fp)
    33→        # Response structure: {choices:[{message:{content}}]} or {content}
    34→        if isinstance(data, dict):
    35→            if "choices" in data and data["choices"]:
    36→                msg = data["choices"][0].get("message", {})
    37→                return msg.get("content", "")
    38→            if "content" in data:
    39→                return data["content"]
    40→            return json.dumps(data)
    41→        return str(data)
    42→    finally:
    43→        try:
    44→            os.unlink(out_path)
    45→        except OSError:
    46→            pass

    86→    """Send document text to AI and get structured extraction."""

   374→# ---- Document text extraction (PDF / txt) ----
   375→
   376→def extract_text_from_pdf(file_path: str) -> str:
   377→    """Extract text from a PDF using pdfplumber."""
   378→    try:
   379→        import pdfplumber
   380→        text_parts = []
   381→        with pdfplumber.open(file_path) as pdf:
   382→            for i, page in enumerate(pdf.pages):
   383→                try:
   384→                    page_text = page.extract_text() or ""
   385→                except Exception:
   386→                    page_text = ""
   387→                text_parts.append(page_text)
   388→                if i >= 50:  # cap pages
   389→                    break
   390→        return "\n".join(text_parts)
   391→    except Exception as e:
   392→        return f"[PDF extraction error: {e}]"

   394→
   395→def extract_text_from_file(file_path: str, filename: str) -> str:
   396→    """Extract text from supported file types."""
   397→    lower = filename.lower()
   398→    if lower.endswith(".pdf"):
   399→        return extract_text_from_pdf(file_path)
   400→    if lower.endswith((".txt", ".md", ".csv", ".json", ".log")):
   401→        try:
   402→            with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
   403→                return f.read()
   404→        except Exception as e:
   405→            return f"[Text extraction error: {e}]"
   406→    # For docx, try a simple zip-based extraction
   407→    if lower.endswith(".docx"):
   408→        try:
   409→            import zipfile, re
   410→            with zipfile.ZipFile(file_path) as z:
   411→                with z.open("word/document.xml") as f:
   412→                    content = f.read().decode("utf-8", errors="ignore")
   413→            # Strip XML tags
   414→            text = re.sub(r"<[^>]+>", " ", content)
   415→            text = re.sub(r"\s+", " ", text)
   416→            return text
   417→        except Exception as e:
   418→            return f"[DOCX extraction error: {e}]"
   419→    return f"[Unsupported file type: {filename}]"

   499→    if category:
   500→        parts.append(category)
   501→    if keywords:
   502→        parts.append(keywords)
   503→    parts.append("tender")
   504→    parts.append("open")
   505→    if location:
   506→        parts.append(location)
   507→    query = " ".join(parts)
   508→
   509→    # --- Step 1: Python scrapers (direct fetch from real tender portals) ---
   510→    scraped_results: List[Dict[str, Any]] = []
   511→    sources_used: List[str] = []
   512→    scrape_errors: List[str] = []
   513→
   514→    try:
   515→        from scraper import scrape_all
   516→        scrape_data = scrape_all(category, keywords, location, limit=num_results)
   517→        scraped_results = scrape_data.get("results", [])
   518→        sources_used = scrape_data.get("sources", [])
   519→        scrape_errors = scrape_data.get("errors", [])
   520→    except Exception as e:
   521→        scrape_errors.append(f"scraper module: {str(e)[:100]}")
   522→
   523→    # --- Step 2: AI web search for additional coverage ---
   524→    web_search_results: List[Dict[str, Any]] = []
   525→    if len(scraped_results) < num_results:
   526→        try:
   527→            raw_results = _run_zai_web_search(query, num=num_results)
   528→            if raw_results:
   529→                # Use AI to structure the raw web search results
   530→                raw_text = json.dumps(raw_results, indent=2, default=str)[:20000]
   531→                prompt = f"""Analyze these web search results and extract ONLY actual tender opportunities.
   532→If a result is a news article, blog post, or generic page (not an actual tender listing), EXCLUDE it.
   533→
   534→The user is looking for: {category} {keywords} in {location}
   535→
   536→RAW SEARCH RESULTS (JSON):
   537→{raw_text}
   538→
   539→Return ONLY the JSON array as specified. Exclude non-tender results."""
   540→                try:
   541→                    response = _run_zai_chat(prompt, system=DISCOVER_SYSTEM)
   542→                    response = response.strip()
   543→                    if response.startswith("```"):
   544→                        response = response.split("\n", 1)[1] if "\n" in response else response[3:]
   545→                        if response.endswith("```"):
   546→                            response = response[:-3]
   547→                        response = response.strip()
   548→                    if response.startswith("json"):
   549→                        response = response[4:].strip()
   550→                    first = response.find("[")
   551→                    last = response.rfind("]")
   552→                    if first >= 0 and last > first:
   553→                        response = response[first:last + 1]
   554→                    structured = json.loads(response)
   555→                    if isinstance(structured, list):
   556→                        web_search_results = structured
   557→                        if web_search_results:
   558→                            sources_used.append(f"AI web search ({len(web_search_results)} tenders)")
   559→                except (json.JSONDecodeError, RuntimeError):
   560→                    pass
   561→        except Exception:
   562→            pass
   563→
   564→    # --- Merge & deduplicate ---
   565→    all_results = scraped_results + web_search_results
   566→    seen_urls = set()
   567→    seen_titles = set()
   568→    deduped = []
   569→    for r in all_results:
   570→        url = r.get("url", "")
   571→        title = r.get("title", "").lower().strip()[:80]
   572→        if url in seen_urls or title in seen_titles:
   573→            continue
   574→        seen_urls.add(url)
   575→        seen_titles.add(title)
   576→        deduped.append(r)
   577→
   578→    # Sort by relevance then closing date
   579→    rel_order = {"high": 0, "medium": 1, "low": 2}
   580→    deduped.sort(key=lambda r: (rel_order.get(r.get("relevance", "medium"), 1), r.get("closing_date") or "9999"))
   581→
   582→    return {
   583→        "query": query,
   584→        "results": deduped[:num_results],
   585→        "sources": sources_used,
   586→        "errors": scrape_errors,
   587→        "raw_count": len(deduped),
   588→        "scraped_count": len(scraped_results),
   589→        "web_search_count": len(web_search_results),
   590→    }

   634→def _build_company_header(company: Dict[str, Any]) -> str:
   635→    """Build a company letterhead header string from company profile data."""
   636→    lines = []
   637→    if company.get("company_name"):

   639→    if company.get("address"):
   640→        lines.append(company["address"])
   641→    contact_parts = []

   643→        contact_parts.append(f"Tel: {company['phone']}")
   644→    if company.get("email"):
   645→        contact_parts.append(f"Email: {company['email']}")
   646→    if company.get("website"):
   647→        contact_parts.append(f"Web: {company['website']}")
   648→    if contact_parts:
   649→        lines.append(" | ".join(contact_parts))
   650→    reg_parts = []
   651→    if company.get("registration_number"):
   652→        reg_parts.append(f"Reg No: {company['registration_number']}")
   653→    if company.get("vat_number"):
   654→        reg_parts.append(f"VAT: {company['vat_number']}")
   655→    if company.get("psira_number"):
   656→        reg_parts.append(f"PSIRA: {company['psira_number']}")
   657→    if reg_parts:
   658→        lines.append(" | ".join(reg_parts))
   659→    if company.get("bbbbee_level"):
   660→        lines.append(f"B-BBEE: {company['bbbbee_level']}")
   661→    return "\n".join(lines) if lines else "[COMPANY NAME]"
