    93→// ---- Helpers ----
    94→function formatNumber(n: number): string {
    95→  return new Intl.NumberFormat("en-ZA").format(n);
    96→}
    97→
    98→function deadlineBadgeClass(days: number): string {
    99→  if (days < 0 || days <= 1)
   100→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   101→  if (days <= 3)
   102→    return "bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-900";
   103→  return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-900";
   104→}
   105→
   106→function deadlineLabel(days: number): string {
   107→  if (days < 0) return `Overdue by ${Math.abs(days)}d`;
   108→  if (days === 0) return "Due today";
   109→  if (days === 1) return "1 day left";
   110→  return `${days} days left`;
   111→}
   112→
   113→function docExpiryLabel(days: number): string {
   114→  if (days < 0) return `Expired ${Math.abs(days)}d ago`;
   115→  if (days === 0) return "Expires today";
   116→  if (days === 1) return "Expires tomorrow";
   117→  return `In ${days}d`;
   118→}

   220→// ---- Main Component ----
   221→export function DashboardTab() {
   222→  const [data, setData] = useState<DashboardData | null>(null);
   223→  const [loading, setLoading] = useState(true);
   224→  const [refreshing, setRefreshing] = useState(false);
   225→  const [error, setError] = useState<string | null>(null);
   226→
   227→  const fetchData = useCallback(async (isRefresh = false) => {
   228→    if (isRefresh) setRefreshing(true);
   229→    else setLoading(true);
   230→    setError(null);
   231→    try {
   232→      const d = (await api.dashboard()) as DashboardData;
   233→      setData(d);
   234→    } catch (e: any) {
   235→      setError(e?.message || "Failed to load dashboard data");
   236→    } finally {
   237→      setLoading(false);
   238→      setRefreshing(false);
   239→    }
   240→  }, []);
   241→
   242→  useEffect(() => {
   243→    fetchData();
   244→  }, [fetchData]);
   245→
   246→  const openTender = (id: number) => {
   247→    useDashboardStore.getState().setTab("tenders");
   248→    setTimeout(() => useDashboardStore.getState().selectTender(id), 50);
   249→  };
   250→
   251→  const goToTab = (tab: DashboardTabId) =>
   252→    useDashboardStore.getState().setTab(tab);
   253→
   254→  // Top 8 statuses by count, for the breakdown chart
   255→  const statusRows = useMemo(() => {
   256→    if (!data?.status_counts) return [];

   258→      .filter(([, c]) => c > 0)
   259→      .sort((a, b) => b[1] - a[1])
   260→      .slice(0, 8);
   261→  }, [data]);
   262→
   263→  const statusMax = useMemo(
   264→    () => statusRows.reduce((m, [, c]) => Math.max(m, c), 0) || 1,
   265→    [statusRows]
   266→  );
   267→
   268→  // ---- Loading state ----
   269→  if (loading) {
   270→    return (
   271→      <div className="space-y-6">
   272→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   273→          <div className="space-y-2">
   274→            <Skeleton className="h-8 w-40" />
   275→            <Skeleton className="h-4 w-56" />
   276→          </div>
   277→          <div className="flex gap-2">
   278→            <Skeleton className="h-9 w-24" />
   279→            <Skeleton className="h-9 w-28" />
   280→          </div>
   281→        </div>
   282→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   283→          {Array.from({ length: 4 }).map((_, i) => (
   284→            <KpiSkeleton key={i} />
   285→          ))}
   286→        </div>
   287→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   288→          <Card className="lg:col-span-2">
   289→            <CardHeader>
   290→              <Skeleton className="h-5 w-48" />
   291→              <Skeleton className="h-4 w-64" />
   292→            </CardHeader>
   293→            <CardContent className="space-y-3">
   294→              {Array.from({ length: 5 }).map((_, i) => (
   295→                <Skeleton key={i} className="h-6 w-full" />
   296→              ))}
   297→            </CardContent>
   298→          </Card>
   299→          <Card>
   300→            <CardHeader>
   301→              <Skeleton className="h-5 w-32" />
   302→              <Skeleton className="h-4 w-40" />
   303→            </CardHeader>
   304→            <CardContent className="space-y-3">
   305→              {Array.from({ length: 4 }).map((_, i) => (
   306→                <Skeleton key={i} className="h-8 w-full" />
   307→              ))}
   308→            </CardContent>
   309→          </Card>
   310→        </div>
   311→      </div>
   312→    );
   313→  }
   314→
   315→  // ---- Error state ----
   316→  if (error) {
   317→    return (
   318→      <div className="space-y-6">
   319→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   320→          <div>
   321→            <h1 className="text-2xl font-bold tracking-tight">Dashboard</h1>
   322→            <p className="text-sm text-muted-foreground">
   323→              Tender activity at a glance
   324→            </p>
   325→          </div>
   326→        </div>
   327→        <Alert variant="destructive">
   328→          <AlertTriangle className="h-4 w-4" />
   329→          <AlertTitle>Couldn&apos;t load your dashboard</AlertTitle>
   330→          <AlertDescription>
   331→            <p className="mb-3">{error}</p>
   332→            <Button
   333→              size="sm"
   334→              variant="outline"
   335→              onClick={() => fetchData()}
   336→              className="border-destructive/40 text-destructive hover:bg-destructive/10"
   337→            >
   338→              <RefreshCw className="h-3.5 w-3.5" />
   339→              Try again
   340→            </Button>
   341→          </AlertDescription>
   342→        </Alert>
   343→      </div>
   344→    );
   345→  }
   346→
   347→  if (!data) return null;
   348→
   349→  const docsExpiringTotal = data.docs_expiring_soon + data.docs_expired;
   350→
   351→  return (
   352→    <div className="space-y-6">
   353→      {/* ---- Page header ---- */}
   354→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   355→        <div>
   356→          <h1 className="text-2xl font-bold tracking-tight">Dashboard</h1>
   357→          <p className="text-sm text-muted-foreground">
   358→            Tender activity at a glance
   359→          </p>
   360→        </div>
   361→        <div className="flex items-center gap-2">
   362→          <Button
   363→            variant="ghost"
   364→            onClick={() => fetchData(true)}
   365→            disabled={refreshing}
   366→          >
   367→            <RefreshCw
   368→              className={cn("h-4 w-4", refreshing && "animate-spin")}
   369→            />
   370→            Refresh
   371→          </Button>
   372→          <Button onClick={() => goToTab("tenders")}>
   373→            <Plus className="h-4 w-4" />
   374→            New Tender
   375→          </Button>
   376→        </div>
   377→      </div>
   378→
   379→      {/* ---- KPI cards ---- */}
   380→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   381→        <KpiCard
   382→          icon={FileStack}
   383→          iconClass="bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400"
   384→          value={data.active_count}
   385→          label="Active Tenders"
   386→          subtitle="In progress"
   387→        />
   388→        <KpiCard
   389→          icon={CalendarClock}
   390→          iconClass="bg-amber-100 text-amber-600 dark:bg-amber-950 dark:text-amber-400"
   391→          value={data.closing_this_week}
   392→          label="Closing This Week"
   393→          subtitle="Urgent"
   394→        />
   395→        <KpiCard
   396→          icon={FolderClock}
   397→          iconClass="bg-rose-100 text-rose-600 dark:bg-rose-950 dark:text-rose-400"
   398→          value={docsExpiringTotal}
   399→          label="Documents Expiring"
   400→          subtitle="Need attention"
   401→        />
   402→        <KpiCard
   403→          icon={Trophy}
   404→          iconClass="bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400"
   405→          value={data.awarded_this_year}
   406→          label="Awarded This Year"
   407→          subtitle="Wins"
   408→        />
   409→      </div>
   410→
   411→      {/* ---- Three-column section ---- */}
   412→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   413→        {/* Left column (span 2) */}
   414→        <div className="space-y-4 lg:col-span-2">
   415→          {/* Tender Status Breakdown */}
   416→          <Card>
   417→            <CardHeader>
   418→              <CardTitle className="flex items-center gap-2">
   419→                <FileStack className="h-4 w-4 text-primary" />
   420→                Tender Status Breakdown
   421→              </CardTitle>
   422→              <CardDescription>
   423→                Distribution across {formatNumber(data.total_tenders)} tender
   424→                {data.total_tenders === 1 ? "" : "s"}
   425→              </CardDescription>
   426→            </CardHeader>
   427→            <CardContent className="space-y-2.5">
   428→              {statusRows.length === 0 ? (
   429→                <EmptyState
   430→                  icon={FileStack}
   431→                  message="No tenders yet. Create your first tender to see a status breakdown here."
   432→                />
   433→              ) : (
   434→                statusRows.map(([status, count]) => (
   435→                  <div
   436→                    key={status}
   437→                    className="flex items-center gap-3 text-sm"
   438→                  >
   439→                    <div className="w-40 shrink-0 truncate font-medium text-foreground">
   440→                      {status}
   441→                    </div>
   442→                    <div className="relative h-6 flex-1 overflow-hidden rounded-md bg-muted">
   443→                      <div
   444→                        className={cn(
   445→                          "h-full rounded-md transition-all",
   446→                          statusBarColor(status)
   447→                        )}
   448→                        style={{
   449→                          width: `${Math.max(
   450→                            6,
   451→                            Math.round((count / statusMax) * 100)
   452→                          )}%`,
   453→                        }}
   454→                      />
   455→                    </div>
   456→                    <div className="w-10 shrink-0 text-right tabular-nums font-medium text-foreground">
   457→                      {formatNumber(count)}
   458→                    </div>
   459→                  </div>
   460→                ))
   461→              )}
   462→            </CardContent>
   463→          </Card>
   464→
   465→          {/* Tenders Closing This Week */}
   466→          <Card>
   467→            <CardHeader>
   468→              <CardTitle className="flex items-center gap-2">
   469→                <CalendarClock className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   470→                Tenders Closing This Week
   471→              </CardTitle>
   472→              <CardDescription>
   473→                {data.closing_this_week > 0
   474→                  ? `${formatNumber(data.closing_this_week)} tender${
   475→                      data.closing_this_week === 1 ? "" : "s"
   476→                    } need attention`
   477→                  : "All clear for this week"}
   478→              </CardDescription>
   479→            </CardHeader>
   480→            <CardContent>
   481→              {data.closing_this_week_list.length === 0 ? (
   482→                <EmptyState
   483→                  icon={CheckCircle2}
   484→                  message="No tenders closing this week — you're on top of things."
   485→                  iconClass="bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400"
   486→                />
   487→              ) : (
   488→                <div className="max-h-80 space-y-1 overflow-y-auto scrollbar-thin pr-1">
   489→                  {data.closing_this_week_list.map((t) => (
   490→                    <button
   491→                      key={t.id}
   492→                      onClick={() => openTender(t.id)}
   493→                      className="group w-full rounded-lg border border-transparent px-3 py-2.5 text-left transition-colors hover:border-border hover:bg-accent/50"
   494→                    >
   495→                      <div className="flex items-start justify-between gap-3">
   496→                        <div className="min-w-0 flex-1">
   497→                          <div className="flex items-center gap-2">
   498→                            <span className="font-mono text-xs font-semibold text-primary">
   499→                              {t.reference_number}
   500→                            </span>
   501→                            <span
   502→                              className={cn(
   503→                                "inline-flex items-center rounded-md px-1.5 py-0.5 text-[10px] font-medium",
   504→                                statusColor(t.status)
   505→                              )}
   506→                            >
   507→                              {t.status}
   508→                            </span>
   509→                          </div>
   510→                          <div className="mt-0.5 truncate text-sm font-medium text-foreground">
   511→                            {t.title}
   512→                          </div>
   513→                          <div className="mt-0.5 truncate text-xs text-muted-foreground">
   514→                            {t.client_company || "—"} · Closes{" "}

   517→                        </div>

   519→                          variant="outline"
   520→                          className={cn(
   521→                            "shrink-0 tabular-nums",
   522→                            deadlineBadgeClass(t.days_remaining)
   523→                          )}
   524→                        >
   525→                          <Clock className="h-3 w-3" />
   526→                          {deadlineLabel(t.days_remaining)}
   527→                        </Badge>
   528→                      </div>
   529→                    </button>
   530→                  ))}
   531→                </div>
   532→              )}
   533→            </CardContent>
   534→          </Card>
   535→
   536→          {/* Awarded This Year */}
   537→          <Card>
   538→            <CardHeader>
   539→              <CardTitle className="flex items-center gap-2">
   540→                <Trophy className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   541→                Awarded This Year
   542→              </CardTitle>
   543→              <CardDescription>
   544→                {data.awarded_this_year > 0
   545→                  ? `${formatNumber(data.awarded_this_year)} win${
   546→                      data.awarded_this_year === 1 ? "" : "s"
   547→                    } this year`
   548→                  : "Submit tenders to start winning"}
   549→              </CardDescription>
   550→            </CardHeader>
   551→            <CardContent>

   558→              ) : (
   559→                <div className="space-y-1">
   560→                  {data.awarded_this_year_list.map((t) => (
   561→                    <button
   562→                      key={t.id}
   563→                      onClick={() => openTender(t.id)}
   564→                      className="group flex w-full items-center justify-between gap-3 rounded-lg border border-transparent px-3 py-2.5 text-left transition-colors hover:border-border hover:bg-accent/50"
   565→                    >
   566→                      <div className="min-w-0 flex-1">
   567→                        <div className="flex items-center gap-2">
   568→                          <span className="font-mono text-xs font-semibold text-primary">
   569→                            {t.reference_number}
   570→                          </span>
   571→                        </div>
   572→                        <div className="mt-0.5 truncate text-sm font-medium text-foreground">
   573→                          {t.title}
   574→                        </div>
