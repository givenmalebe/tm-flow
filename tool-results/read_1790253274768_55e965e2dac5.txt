   131→function typeBadgeClass(type: BriefingType): string {
   132→  return type === "Briefing"

   137→function dateHeading(iso: string): { label: string; relative: string | null; diff: number } {
   138→  const d = new Date(iso);
   139→  if (isNaN(d.getTime())) return { label: formatDate(iso), relative: null, diff: 0 };
   140→  const today = new Date();
   141→  today.setHours(0, 0, 0, 0);
   142→  d.setHours(0, 0, 0, 0);
   143→  const diff = Math.round((d.getTime() - today.getTime()) / 86400000);
   144→  const label = d.toLocaleDateString("en-ZA", {
   145→    weekday: "short",
   146→    day: "2-digit",
   147→    month: "short",
   148→    year: "numeric",
   149→  });
   150→  let relative: string | null = null;
   151→  if (diff === 0) relative = "Today";
   152→  else if (diff === 1) relative = "Tomorrow";
   153→  else if (diff === -1) relative = "Yesterday";
   154→  else if (diff > 1 && diff <= 7) relative = `In ${diff} days`;
   155→  else if (diff < 0) relative = `${Math.abs(diff)}d ago`;
   156→  return { label, relative, diff };
   157→}
   158→
   159→function relativeBadgeClass(diff: number): string {
   160→  if (diff < 0)
   161→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   162→  if (diff <= 1)
   163→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   164→  if (diff <= 3)
   165→    return "bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-900";
   166→  return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-900";
   167→}

   455→// ---- Briefing dialog (create / edit) ----
   456→interface BriefingDialogProps {
   457→  open: boolean;
   458→  editing: Briefing | null;

   460→  tendersLoading: boolean;

   464→
   465→function BriefingDialog({
   466→  open,
   467→  editing,
   468→  tenders,
   469→  tendersLoading,
   470→  onClose,
   471→  onSaved,
   472→}: BriefingDialogProps) {
   473→  const [form, setForm] = useState<BriefingForm>(EMPTY_FORM);
   474→  const [saving, setSaving] = useState(false);
   475→  const isEdit = editing !== null;
   476→
   477→  useEffect(() => {
   478→    if (open) {
   479→      if (editing) {
   480→        setForm({
   481→          tender_id: String(editing.tender_id),
   482→          briefing_type: editing.briefing_type || "Briefing",
   483→          date: editing.date || "",
   484→          time: editing.time || "",
   485→          location: editing.location || "",
   486→          contact_person: editing.contact_person || "",
   487→          attendance_required: editing.attendance_required,
   488→          attendance_confirmed: editing.attendance_confirmed,
   489→          person_attending: editing.person_attending || "",
   490→          notes: editing.notes || "",
   491→        });
   492→      } else {
   493→        setForm(EMPTY_FORM);
   494→      }
   495→    }
   496→  }, [open, editing]);
   497→
   498→  const update = <K extends keyof BriefingForm>(key: K, value: BriefingForm[K]) =>
   499→    setForm((f) => ({ ...f, [key]: value }));
   500→
   501→  const canSave =
   502→    form.date.length > 0 && (isEdit || form.tender_id.length > 0);
   503→
   504→  const handleSave = async () => {
   505→    if (!form.date) {
   506→      toast.error("Date is required");
   507→      return;
   508→    }
   509→    if (!isEdit && !form.tender_id) {
   510→      toast.error("Please select a tender");
   511→      return;
   512→    }
   513→    setSaving(true);
   514→    try {
   515→      const payload: Record<string, any> = {
   516→        briefing_type: form.briefing_type,
   517→        date: form.date,
   518→        time: form.time || null,
   519→        location: form.location || null,
   520→        contact_person: form.contact_person || null,
   521→        attendance_required: form.attendance_required,
   522→        attendance_confirmed: form.attendance_confirmed,
   523→        person_attending: form.person_attending || null,
   524→        notes: form.notes || null,
   525→      };
   526→      if (isEdit && editing) {
   527→        await api.updateBriefing(editing.id, payload);
   528→        toast.success("Briefing updated");
   529→      } else {
   530→        const tenderId = Number(form.tender_id);
   531→        await api.createBriefing(tenderId, { ...payload, tender_id: tenderId });
   532→        toast.success("Briefing added");
   533→      }
   534→      onSaved();
   535→      onClose();
   536→    } catch (e: any) {
   537→      toast.error(e?.message || "Failed to save briefing");
   538→    } finally {
   539→      setSaving(false);
   540→    }
   541→  };
   542→
   543→  const selectedTender = tenders.find((t) => String(t.id) === form.tender_id);
   544→
   545→  return (
   546→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   547→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto scrollbar-thin">
   548→        <DialogHeader>
   549→          <DialogTitle className="flex items-center gap-2">
   550→            {isEdit ? (
   551→              <>
   552→                <Pencil className="h-4 w-4 text-primary" />
   553→                Edit Briefing
   554→              </>
   555→            ) : (
   556→              <>
   557→                <CalendarPlus className="h-4 w-4 text-primary" />
   558→                Add Briefing
   559→              </>
   560→            )}
   561→          </DialogTitle>
   562→          <DialogDescription>
   563→            {isEdit
   564→              ? "Update the briefing or site inspection details."
   565→              : "Schedule a compulsory briefing or site inspection for a tender."}
   566→          </DialogDescription>
   567→        </DialogHeader>
   568→
   569→        <div className="grid gap-4 py-2">
   570→          {/* Tender selector */}
   571→          <div className="grid gap-2">
   572→            <Label htmlFor="b-tender">
   573→              Tender <span className="text-rose-500">*</span>
   574→            </Label>
   575→            {isEdit ? (
   576→              <div className="flex items-center gap-2 rounded-md border bg-muted/40 px-3 py-2 text-sm">
   577→                <MapPin className="h-4 w-4 shrink-0 text-muted-foreground" />
   578→                <span className="truncate text-muted-foreground">
   579→                  <span className="font-mono text-xs font-semibold text-foreground">
   580→                    {editing?.tender_reference}
   581→                  </span>{" "}
   582→                  — {editing?.tender_title}
   583→                </span>
