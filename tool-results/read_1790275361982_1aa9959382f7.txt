   185→function toDatetimeLocalValue(iso: string | null): string {
   186→  if (!iso) return "";
   187→  try {
   188→    const d = new Date(iso);
   189→    if (isNaN(d.getTime())) return "";
   190→    const pad = (n: number) => String(n).padStart(2, "0");
   191→    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(
   192→      d.getHours()
   193→    )}:${pad(d.getMinutes())}`;
   194→  } catch {
   195→    return "";
   196→  }
   197→}

   488→// ---- Main component ----
   489→export function ClarificationsTab() {
   490→  const [tenders, setTenders] = useState<TenderOption[]>([]);
   491→  const [communications, setCommunications] = useState<FlattenedCommunication[]>([]);
   492→  const [loading, setLoading] = useState(true);
   493→  const [refreshing, setRefreshing] = useState(false);
   494→  const [error, setError] = useState<string | null>(null);
   495→
   496→  const [typeFilter, setTypeFilter] = useState<string>("all");
   497→  const [search, setSearch] = useState("");
   498→
   499→  const [dialogOpen, setDialogOpen] = useState(false);
   500→  const [form, setForm] = useState<CommForm>(EMPTY_FORM);
   501→  const [saving, setSaving] = useState(false);
   502→
   503→  const fetchTenders = useCallback(async (isRefresh = false) => {
   504→    if (isRefresh) setRefreshing(true);
   505→    else setLoading(true);
   506→    setError(null);
   507→    try {
   508→      const res = await api.listTenders({ sort: "updated_at", order: "desc" });
   509→      const items = (res?.items || []) as TenderOption[];
   510→      setTenders(items);
   511→
   512→      const flat: FlattenedCommunication[] = [];
   513→      for (const t of items) {
   514→        const tender = t as any;
   515→        const comms = (tender.communications || []) as CommunicationRaw[];
   516→        for (const c of comms) {
   517→          flat.push({
   518→            ...c,
   519→            tender_id: tender.id,
   520→            tender_reference: tender.reference_number || "—",
   521→            tender_title: tender.title || "",
   522→          });
   523→        }
   524→      }
   525→      // Newest first
   526→      flat.sort((a, b) => {
   527→        const da = a.comm_date ? new Date(a.comm_date).getTime() : 0;
   528→        const db = b.comm_date ? new Date(b.comm_date).getTime() : 0;
   529→        if (db !== da) return db - da;
   530→        return b.id - a.id;
   531→      });
   532→      setCommunications(flat);
   533→    } catch (e: any) {
   534→      setError(e?.message || "Failed to load communications");
   535→    } finally {
   536→      setLoading(false);
   537→      setRefreshing(false);
   538→    }
   539→  }, []);
   540→
   541→  useEffect(() => {
   542→    fetchTenders();
   543→  }, [fetchTenders]);
   544→
   545→  const openTender = (id: number) => {
   546→    useDashboardStore.getState().setTab("tenders");
   547→    setTimeout(() => useDashboardStore.getState().selectTender(id), 50);
   548→  };
   549→
   550→  // ---- Stats ----
   551→  const stats = useMemo(() => {
   552→    const total = communications.length;
   553→    const clarificationRequests = communications.filter(
   554→      (c) => c.comm_type === "Clarification Request"
   555→    ).length;
   556→    const meetings = communications.filter((c) => c.comm_type === "Meeting").length;
   557→    const extensions = communications.filter(
   558→      (c) =>
   559→        c.comm_type === "Extension" ||
   560→        c.comm_type === "Tender Validity Extension"
   561→    ).length;
   562→    return { total, clarificationRequests, meetings, extensions };
   563→  }, [communications]);
   564→
   565→  // ---- Filtered communications ----
   566→  const filtered = useMemo(() => {
   567→    const q = search.trim().toLowerCase();
   568→    return communications.filter((c) => {
   569→      if (typeFilter !== "all" && c.comm_type !== typeFilter) return false;
   570→      if (!q) return true;
   571→      const hay = [
   572→        c.subject || "",
   573→        c.body || "",
   574→        c.tender_reference || "",
   575→        c.tender_title || "",
   576→        c.from_party || "",
   577→        c.to_party || "",
   578→      ]
   579→        .join(" ")
   580→        .toLowerCase();
   581→      return hay.includes(q);
   582→    });
   583→  }, [communications, typeFilter, search]);
   584→
   585→  // ---- Create communication ----
   586→  const handleSave = async () => {
   587→    if (!form.tender_id) {
   588→      toast.error("Please select a tender");
   589→      return;
   590→    }
   591→    if (!form.comm_type) {
   592→      toast.error("Please select a communication type");
   593→      return;
   594→    }
   595→    setSaving(true);
   596→    try {
   597→      const tenderId = Number(form.tender_id);
   598→      await api.createCommunication(tenderId, {
   599→        tender_id: tenderId,
