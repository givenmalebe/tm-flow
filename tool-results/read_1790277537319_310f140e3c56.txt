   102→// ---- Helpers ----
   103→function formatNumber(n: number): string {
   104→  return new Intl.NumberFormat("en-ZA").format(n);
   105→}
   106→
   107→function deadlineBadgeClass(days: number): string {
   108→  if (days < 0 || days <= 1)
   109→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   110→  if (days <= 3)
   111→    return "bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-900";
   112→  return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-900";
   113→}
   114→
   115→function deadlineLabel(days: number): string {
   116→  if (days < 0) return `Overdue by ${Math.abs(days)}d`;
   117→  if (days === 0) return "Due today";
   118→  if (days === 1) return "1 day left";
   119→  return `${days} days left`;
   120→}
   121→
   122→function docExpiryLabel(days: number): string {
   123→  if (days < 0) return `Expired ${Math.abs(days)}d ago`;
   124→  if (days === 0) return "Expires today";
   125→  if (days === 1) return "Expires tomorrow";
   126→  return `In ${days}d`;
   127→}

   156→// ---- KPI Card ----
   157→interface KpiCardProps {
   158→  icon: LucideIcon;
   159→  iconClass: string;
   160→  value: number;
   161→  label: string;
   162→  subtitle: string;
   163→  onClick?: () => void;

   166→
   167→function KpiCard({ icon: Icon, iconClass, value, label, subtitle, onClick, hint }: KpiCardProps) {
   168→  const clickable = !!onClick;
   169→  const CardTag = clickable ? "button" : "div";
   170→  return (
   171→    <Card
   172→      asChild={clickable}
   173→      className={cn(
   174→        "text-left transition-all",
   175→        clickable
   176→          ? "cursor-pointer hover:shadow-md hover:border-primary/40 hover:-translate-y-0.5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
   177→          : ""
   178→      )}
   179→    >
   180→      <CardTag onClick={onClick} type={clickable ? "button" : undefined}>
   181→        <CardContent className="flex items-start justify-between gap-3 p-5">
   182→          <div className="min-w-0">
   183→            <div className="text-3xl font-bold tracking-tight tabular-nums text-foreground">
   184→              {formatNumber(value)}
   185→            </div>
   186→            <div className="mt-1 text-sm font-medium text-foreground">{label}</div>
   187→            <div className="text-xs text-muted-foreground mt-0.5">{subtitle}</div>
   188→            {clickable && (
   189→              <div className="mt-2 inline-flex items-center gap-0.5 text-[11px] font-medium text-primary">
   190→                {hint || "Click to view"}
   191→                <ChevronRight className="h-3 w-3" />
   192→              </div>
   193→            )}
   194→          </div>
   195→          <div
   196→            className={cn(
   197→              "flex h-11 w-11 shrink-0 items-center justify-center rounded-xl",
   198→              iconClass
   199→            )}
   200→          >
   201→            <Icon className="h-5 w-5" />
   202→          </div>
   203→        </CardContent>
   204→      </CardTag>
   205→    </Card>
   206→  );
   207→}

   250→export function DashboardTab() {

   254→  const [error, setError] = useState<string | null>(null);
   255→  const [showClosingWeek, setShowClosingWeek] = useState(false);
   256→  const [showAwarded, setShowAwarded] = useState(false);
   257→
   258→  const fetchData = useCallback(async (isRefresh = false) => {
   259→    if (isRefresh) setRefreshing(true);
   260→    else setLoading(true);
   261→    setError(null);

   263→      const d = (await api.dashboard()) as DashboardData;
   264→      setData(d);
   265→    } catch (e: any) {

   268→      setLoading(false);

   270→    }
   271→  }, []);
   272→
   273→  useEffect(() => {
   274→    fetchData();
   275→  }, [fetchData]);
   276→
   277→  const openTender = (id: number) => {
   278→    useDashboardStore.getState().setTab("tenders");
   279→    setTimeout(() => useDashboardStore.getState().selectTender(id), 50);
   280→  };
   281→
   282→  const goToTab = (tab: DashboardTabId) =>
   283→    useDashboardStore.getState().setTab(tab);
   284→
   285→  // Top 8 statuses by count, for the breakdown chart

   287→    if (!data?.status_counts) return [];

   290→      .sort((a, b) => b[1] - a[1])

   292→  }, [data]);
   293→
   294→  const statusMax = useMemo(
   295→    () => statusRows.reduce((m, [, c]) => Math.max(m, c), 0) || 1,
   296→    [statusRows]
   297→  );
   298→
   299→  // ---- Loading state ----
   300→  if (loading) {
   301→    return (
   302→      <div className="space-y-6">
   303→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   304→          <div className="space-y-2">
   305→            <Skeleton className="h-8 w-40" />
   306→            <Skeleton className="h-4 w-56" />
   307→          </div>
   308→          <div className="flex gap-2">
   309→            <Skeleton className="h-9 w-24" />
   310→            <Skeleton className="h-9 w-28" />
   311→          </div>
   312→        </div>
   313→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   314→          {Array.from({ length: 4 }).map((_, i) => (
   315→            <KpiSkeleton key={i} />
   316→          ))}
   317→        </div>
   318→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   319→          <Card className="lg:col-span-2">
   320→            <CardHeader>
   321→              <Skeleton className="h-5 w-48" />
   322→              <Skeleton className="h-4 w-64" />
   323→            </CardHeader>
   324→            <CardContent className="space-y-3">
   325→              {Array.from({ length: 5 }).map((_, i) => (
   326→                <Skeleton key={i} className="h-6 w-full" />
   327→              ))}
   328→            </CardContent>
   329→          </Card>
   330→          <Card>
   331→            <CardHeader>
   332→              <Skeleton className="h-5 w-32" />
   333→              <Skeleton className="h-4 w-40" />
   334→            </CardHeader>
   335→            <CardContent className="space-y-3">
   336→              {Array.from({ length: 4 }).map((_, i) => (
   337→                <Skeleton key={i} className="h-8 w-full" />
   338→              ))}
   339→            </CardContent>
   340→          </Card>
   341→        </div>
   342→      </div>
   343→    );
   344→  }
   345→
   346→  // ---- Error state ----
   347→  if (error) {
   348→    return (
   349→      <div className="space-y-6">
   350→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   351→          <div>
   352→            <h1 className="text-2xl font-bold tracking-tight">Dashboard</h1>
   353→            <p className="text-sm text-muted-foreground">
   354→              Tender activity at a glance
   355→            </p>
   356→          </div>
   357→        </div>
   358→        <Alert variant="destructive">
   359→          <AlertTriangle className="h-4 w-4" />
   360→          <AlertTitle>Couldn&apos;t load your dashboard</AlertTitle>
   361→          <AlertDescription>
   362→            <p className="mb-3">{error}</p>
   363→            <Button
   364→              size="sm"
   365→              variant="outline"
   366→              onClick={() => fetchData()}
   367→              className="border-destructive/40 text-destructive hover:bg-destructive/10"
   368→            >
   369→              <RefreshCw className="h-3.5 w-3.5" />
   370→              Try again
   371→            </Button>
   372→          </AlertDescription>
   373→        </Alert>
   374→      </div>
   375→    );
   376→  }
   377→
   378→  if (!data) return null;
   379→
   380→  const docsExpiringTotal = data.docs_expiring_soon + data.docs_expired;
   381→
   382→  return (
   383→    <div className="space-y-6">
   384→      {/* ---- Page header ---- */}
   385→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   386→        <div>
   387→          <h1 className="text-2xl font-bold tracking-tight">Dashboard</h1>
   388→          <p className="text-sm text-muted-foreground">
   389→            Tender activity at a glance
   390→          </p>
   391→        </div>
   392→        <div className="flex items-center gap-2">
   393→          <Button
   394→            variant="ghost"
   395→            onClick={() => fetchData(true)}
   396→            disabled={refreshing}
   397→          >
   398→            <RefreshCw
   399→              className={cn("h-4 w-4", refreshing && "animate-spin")}
   400→            />
   401→            Refresh
   402→          </Button>
   403→          <Button onClick={() => goToTab("tenders")}>
   404→            <Plus className="h-4 w-4" />
   405→            New Tender
   406→          </Button>
   407→        </div>
   408→      </div>
   409→
   410→      {/* ---- KPI cards ---- */}
   411→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   412→        <KpiCard
   413→          icon={FileStack}
   414→          iconClass="bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400"
   415→          value={data.active_count}
   416→          label="Active Tenders"
   417→          subtitle="In progress"
   418→          onClick={() => goToTab("tenders")}
   419→          hint="View all tenders"
   420→        />
   421→        <KpiCard
   422→          icon={CalendarClock}
   423→          iconClass="bg-amber-100 text-amber-600 dark:bg-amber-950 dark:text-amber-400"
   424→          value={data.closing_this_week}
   425→          label="Closing This Week"
   426→          subtitle="Urgent"
   427→          onClick={() => setShowClosingWeek(true)}
   428→          hint="See which tenders"
   429→        />
   430→        <KpiCard
   431→          icon={FolderClock}
   432→          iconClass="bg-rose-100 text-rose-600 dark:bg-rose-950 dark:text-rose-400"

   434→          label="Documents Expiring"
   435→          subtitle="Need attention"
   436→          onClick={() => goToTab("library")}
   437→          hint="Open compliance library"
   438→        />
   439→        <KpiCard
   440→          icon={Trophy}
   441→          iconClass="bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400"
   442→          value={data.awarded_this_year}

   447→        />
   448→      </div>
   449→
   450→      {/* ---- Three-column section ---- */}

   476→                    className="flex items-center gap-3 text-sm"

   478→                    <div className="w-40 shrink-0 truncate font-medium text-foreground">

   516→                  : "All clear for this week"}
   517→              </CardDescription>
   518→            </CardHeader>
   519→            <CardContent>
   520→              {data.closing_this_week_list.length === 0 ? (

   538→                              {t.reference_number}
   539→                            </span>

   541→                              className={cn(
   542→                                "inline-flex items-center rounded-md px-1.5 py-0.5 text-[10px] font-medium",

   556→                        </div>
   557→                        <Badge
   558→                          variant="outline"
   559→                          className={cn(
   560→                            "shrink-0 tabular-nums",
   561→                            deadlineBadgeClass(t.days_remaining)
   562→                          )}
   563→                        >
   564→                          <Clock className="h-3 w-3" />
   565→                          {deadlineLabel(t.days_remaining)}
   566→                        </Badge>
   567→                      </div>
   568→                    </button>
   569→                  ))}
   570→                </div>
   571→              )}
   572→            </CardContent>
   573→          </Card>
   574→
   575→          {/* Awarded This Year */}
   576→          <Card>
   577→            <CardHeader>
   578→              <CardTitle className="flex items-center gap-2">
   579→                <Trophy className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   580→                Awarded This Year
