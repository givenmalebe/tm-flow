   465→        return None
   466→    try:
   467→        return datetime.strptime(s, "%H:%M").time()

   481→def _save_upload(upload: UploadFile, subdir: str = "") -> str:
   482→    """Save an uploaded file to UPLOAD_DIR/subdir and return relative path."""
   483→    target_dir = os.path.join(UPLOAD_DIR, subdir) if subdir else UPLOAD_DIR
   484→    os.makedirs(target_dir, exist_ok=True)
   485→    safe_name = f"{uuid.uuid4().hex}_{Path(upload.filename).name}"
   486→    target_path = os.path.join(target_dir, safe_name)
   487→    with open(target_path, "wb") as out:
   488→        shutil.copyfileobj(upload.file, out)
   489→    return os.path.relpath(target_path, UPLOAD_DIR)

   498→def _auto_link_library_doc(item: ChecklistItem, db: Session):
   499→    """
   500→    Auto-link a checklist item to a matching compliance library document.
   501→    Sets status to Complete if a valid library doc exists, Expired if the doc is expired.
   502→    """
   503→    if not item.library_category:
   504→        return
   505→    # Find matching library docs by category
   506→    today = date.today()
   507→    matching_docs = db.query(LibraryDocument).filter(
   508→        LibraryDocument.category == item.library_category
   509→    ).all()
   510→    if not matching_docs:
   511→        return
   512→    # Prefer non-expired docs, then the most recently uploaded
   513→    best_doc = None
   514→    best_is_expired = False
   515→    for d in matching_docs:
   516→        if d.expiry_date:
   517→            is_expired = d.expiry_date < today
   518→            if not is_expired and not best_is_expired:
   519→                best_doc = d
   520→                best_is_expired = False
   521→            elif is_expired and best_doc is None:
   522→                best_doc = d
   523→                best_is_expired = True
   524→        else:
   525→            # No expiry = always valid
   526→            if not best_is_expired:
   527→                best_doc = d
   528→                best_is_expired = False
   529→    if best_doc:
   530→        item.linked_library_doc_id = best_doc.id
   531→        item.status = "Expired" if best_is_expired else "Complete"

   534→def _checklist_from_extracted(tender_id: int, extracted: dict, db: Session):
   535→    """
   536→    Create a tender-specific checklist using AI generation.
   537→    Falls back to default categories + compulsory documents if AI generation fails.
   538→    Auto-links each item to a matching compliance library document and sets status.
   539→    """
   540→    # Try AI-generated checklist first
   541→    ai_items = []
   542→    if extracted and not extracted.get("_error"):
   543→        try:
   544→            ai_items = ai_module.generate_tender_checklist(extracted)
   545→        except Exception:
   546→            ai_items = []
   547→
   548→    if ai_items:
   549→        # Use AI-generated items
   550→        for item_data in ai_items:
   551→            item = ChecklistItem(
   552→                tender_id=tender_id,
   553→                name=item_data["name"],
   554→                description=item_data.get("description"),
   555→                category=item_data.get("category", "Other"),
   556→                mandatory=item_data.get("mandatory", True),
   557→                library_category=item_data.get("library_category"),
   558→                status="Outstanding",
   559→            )
   560→            _auto_link_library_doc(item, db)
   561→            db.add(item)
   562→    else:
   563→        # Fallback: default categories + compulsory documents + suggestions
   564→        compulsory = extracted.get("compulsory_documents", []) if extracted else []
   565→        suggestions = extracted.get("checklist_suggestions", []) if extracted else []
   566→        combined = list(dict.fromkeys(CHECKLIST_CATEGORIES + compulsory + suggestions))
   567→        for name in combined:
   568→            item = ChecklistItem(
   569→                tender_id=tender_id,
   570→                name=name[:500],
   571→                status="Outstanding",
   572→                category="Compliance",
   573→                mandatory=True,
   574→            )
   575→            # Try to match a library category by name
   576→            name_lower = name.lower()
   577→            lib_cat_map = {

   702→@app.get("/api/tenders")
   703→def list_tenders(
   704→    search: Optional[str] = None,
   705→    status: Optional[str] = None,
   706→    category: Optional[str] = None,
   707→    person_responsible: Optional[str] = None,
   708→    outcome: Optional[str] = None,
   709→    location: Optional[str] = None,
   710→    closing_date_from: Optional[str] = None,
   711→    closing_date_to: Optional[str] = None,
   712→    value_min: Optional[float] = None,
   713→    value_max: Optional[float] = None,
   714→    submission_date_from: Optional[str] = None,
   715→    submission_date_to: Optional[str] = None,
   716→    sort: str = "updated_at",
   717→    order: str = "desc",
   718→    skip: int = 0,
   719→    limit: int = 200,
   720→    db: Session = Depends(get_db),
   721→    user: User = Depends(get_current_user),
   722→):
   723→    q = db.query(Tender)
   724→    # Text search across multiple fields
   725→    if search:
   726→        like = f"%{search}%"
   727→        q = q.filter(or_(
   728→            Tender.reference_number.ilike(like),
   729→            Tender.title.ilike(like),
   730→            Tender.client_company.ilike(like),
   731→            Tender.location_site.ilike(like),
   732→            Tender.person_responsible.ilike(like),
   733→        ))
   734→    # Exact filters
   735→    if status:
   736→        q = q.filter(Tender.status == status)
   737→    if category:
   738→        q = q.filter(Tender.category == category)
   739→    if person_responsible:
   740→        q = q.filter(Tender.person_responsible == person_responsible)
   741→    if outcome:
   742→        q = q.filter(Tender.outcome == outcome)
   743→    # Location filter (partial match)
   744→    if location:
   745→        q = q.filter(Tender.location_site.ilike(f"%{location}%"))
   746→    # Closing date range
   747→    if closing_date_from:
   748→        try:
   749→            q = q.filter(Tender.closing_date >= date.fromisoformat(closing_date_from))
   750→        except ValueError:
   751→            pass
   752→    if closing_date_to:
   753→        try:
   754→            q = q.filter(Tender.closing_date <= date.fromisoformat(closing_date_to))
   755→        except ValueError:
   756→            pass
   757→    # Value range
   758→    if value_min is not None:
   759→        q = q.filter(Tender.estimated_value >= value_min)
   760→    if value_max is not None:
   761→        q = q.filter(Tender.estimated_value <= value_max)
   762→    # Submission date range (filter tenders that have a submission)
   763→    if submission_date_from or submission_date_to:
   764→        from sqlalchemy import select as sa_select
   765→        # Join with Submission table
   766→        q = q.outerjoin(Submission, Submission.tender_id == Tender.id)
   767→        if submission_date_from:
   768→            try:
   769→                q = q.filter(Submission.submission_date >= date.fromisoformat(submission_date_from))
   770→            except ValueError:
   771→                pass
   772→        if submission_date_to:
   773→            try:
   774→                q = q.filter(Submission.submission_date <= date.fromisoformat(submission_date_to))
   775→            except ValueError:
   776→                pass
   777→    # Sorting
   778→    sort_col = {
   779→        "updated_at": Tender.updated_at,
   780→        "created_at": Tender.created_at,
   781→        "closing_date": Tender.closing_date,
   782→        "estimated_value": Tender.estimated_value,
   783→        "reference_number": Tender.reference_number,
   784→        "title": Tender.title,
   785→        "client_company": Tender.client_company,
   786→        "location_site": Tender.location_site,
   787→        "category": Tender.category,
   788→        "person_responsible": Tender.person_responsible,
   789→    }.get(sort, Tender.updated_at)
   790→    q = q.order_by(asc(sort_col) if order == "asc" else desc(sort_col))
   791→    q = q.distinct()  # Avoid duplicates from outerjoin
   792→    total = q.count()
   793→    tenders = q.offset(skip).limit(limit).all()
   794→    return {"items": [_serialize_tender(t) for t in tenders], "total": total}

  1606→@app.get("/api/briefings/alerts")
  1607→def get_briefing_alerts(db: Session = Depends(get_db), user: User = Depends(get_current_user)):
  1608→    """
  1609→    Get alerts for upcoming compulsory briefings and site inspections.
  1610→    Returns briefings that:
  1611→    - Are within the next 14 days
  1612→    - Have attendance_required = True
  1613→    - Are either: not confirmed, or confirmed but no person assigned
  1614→    Also returns a summary of upcoming briefings by urgency.
  1615→    """
  1616→    today = date.today()
  1617→    all_briefings = db.query(Briefing).join(Tender, Briefing.tender_id == Tender.id).all()
  1618→
  1619→    alerts = []
  1620→    upcoming = []
  1621→    for b in all_briefings:
  1622→        if not b.date:
  1623→            continue
  1624→        days_until = (b.date - today).days
  1625→        briefing_data = {
  1626→            "id": b.id,
  1627→            "tender_id": b.tender_id,
  1628→            "tender_reference": b.tender.reference_number if b.tender else None,
  1629→            "tender_title": b.tender.title if b.tender else None,
  1630→            "client": b.tender.client_company if b.tender else None,
  1631→            "briefing_type": b.briefing_type,
  1632→            "date": b.date.isoformat(),
  1633→            "time": b.time.strftime("%H:%M") if b.time else None,
  1634→            "location": b.location,
  1635→            "contact_person": b.contact_person,
  1636→            "attendance_required": b.attendance_required,
  1637→            "attendance_confirmed": b.attendance_confirmed,
  1638→            "person_attending": b.person_attending,
  1639→            "certificate_path": b.certificate_path,
  1640→            "certificate_url": f"/uploads/{b.certificate_path}?XTransformPort=8000" if b.certificate_path else None,
  1641→            "notes": b.notes,
  1642→            "days_until": days_until,
  1643→            "is_past": days_until < 0,
  1644→            "is_today": days_until == 0,
  1645→        }
  1646→
  1647→        # Upcoming = within next 30 days (or today)
  1648→        if 0 <= days_until <= 30:
  1649→            briefing_data["urgency"] = (
  1650→                "today" if days_until == 0
  1651→                else "critical" if days_until <= 1
  1652→                else "urgent" if days_until <= 3
  1653→                else "soon" if days_until <= 7
  1654→                else "normal"
  1655→            )
  1656→            upcoming.append(briefing_data)
  1657→            # Alert if compulsory and not confirmed, or no person attending
  1658→            if b.attendance_required and (not b.attendance_confirmed or not b.person_attending):
  1659→                alerts.append(briefing_data)
  1660→
  1661→    # Sort by date ascending
  1662→    upcoming.sort(key=lambda x: x["date"])
  1663→    alerts.sort(key=lambda x: x["date"])
  1664→
  1665→    summary = {
  1666→        "total_upcoming": len(upcoming),
  1667→        "total_alerts": len(alerts),
  1668→        "today_count": len([b for b in upcoming if b["is_today"]]),
  1669→        "critical_count": len([b for b in upcoming if b["urgency"] == "critical"]),
  1670→        "urgent_count": len([b for b in upcoming if b["urgency"] == "urgent"]),
  1671→        "soon_count": len([b for b in upcoming if b["urgency"] == "soon"]),
  1672→        "unconfirmed_count": len([b for b in upcoming if b["attendance_required"] and not b["attendance_confirmed"]]),
  1673→        "no_attendee_count": len([b for b in upcoming if b["attendance_required"] and not b["person_attending"]]),
  1674→    }
  1675→
  1676→    return {
  1677→        "alerts": alerts,
  1678→        "upcoming": upcoming,
  1679→        "summary": summary,
  1680→    }
