    98→    98→function deadlineBadgeClass(days: number): string {
    99→    99→  if (days < 0 || days <= 1)
   100→   100→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   101→   101→  if (days <= 3)
   102→   102→    return "bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-900";
   103→   103→  return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-900";
   104→   104→}

   106→   106→function deadlineLabel(days: number): string {
   107→   107→  if (days < 0) return `Overdue by ${Math.abs(days)}d`;
   108→   108→  if (days === 0) return "Due today";
   109→   109→  if (days === 1) return "1 day left";
   110→   110→  return `${days} days left`;
   111→   111→}
   112→   112→
   113→   113→function docExpiryLabel(days: number): string {
   114→   114→  if (days < 0) return `Expired ${Math.abs(days)}d ago`;
   115→   115→  if (days === 0) return "Expires today";
   116→   116→  if (days === 1) return "Expires tomorrow";
   117→   117→  return `In ${days}d`;
   118→   118→}

   223→   223→  const [loading, setLoading] = useState(true);

   225→   225→  const [error, setError] = useState<string | null>(null);
   226→   226→
   227→   227→  const fetchData = useCallback(async (isRefresh = false) => {

   229→   229→    else setLoading(true);
   230→   230→    setError(null);
   231→   231→    try {
   232→   232→      const d = (await api.dashboard()) as DashboardData;
   233→   233→      setData(d);
   234→   234→    } catch (e: any) {
   235→   235→      setError(e?.message || "Failed to load dashboard data");
   236→   236→    } finally {
   237→   237→      setLoading(false);
   238→   238→      setRefreshing(false);
   239→   239→    }
   240→   240→  }, []);
   241→   241→
   242→   242→  useEffect(() => {
   243→   243→    fetchData();
   244→   244→  }, [fetchData]);
   245→   245→
   246→   246→  const openTender = (id: number) => {
   247→   247→    useDashboardStore.getState().setTab("tenders");
   248→   248→    setTimeout(() => useDashboardStore.getState().selectTender(id), 50);
   249→   249→  };
   250→   250→
   251→   251→  const goToTab = (tab: DashboardTabId) =>
   252→   252→    useDashboardStore.getState().setTab(tab);
   253→   253→
   254→   254→  // Top 8 statuses by count, for the breakdown chart

   258→   258→      .filter(([, c]) => c > 0)
   259→   259→      .sort((a, b) => b[1] - a[1])

   261→   261→  }, [data]);
   262→   262→
   263→   263→  const statusMax = useMemo(
   264→   264→    () => statusRows.reduce((m, [, c]) => Math.max(m, c), 0) || 1,
   265→   265→    [statusRows]
   266→   266→  );
   267→   267→
   268→   268→  // ---- Loading state ----
   269→   269→  if (loading) {
   270→   270→    return (
   271→   271→      <div className="space-y-6">
   272→   272→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   273→   273→          <div className="space-y-2">
   274→   274→            <Skeleton className="h-8 w-40" />
   275→   275→            <Skeleton className="h-4 w-56" />
   276→   276→          </div>
   277→   277→          <div className="flex gap-2">
   278→   278→            <Skeleton className="h-9 w-24" />
   279→   279→            <Skeleton className="h-9 w-28" />
   280→   280→          </div>
   281→   281→        </div>
   282→   282→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   283→   283→          {Array.from({ length: 4 }).map((_, i) => (
   284→   284→            <KpiSkeleton key={i} />
   285→   285→          ))}
   286→   286→        </div>
   287→   287→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   288→   288→          <Card className="lg:col-span-2">
   289→   289→            <CardHeader>
   290→   290→              <Skeleton className="h-5 w-48" />
   291→   291→              <Skeleton className="h-4 w-64" />
   292→   292→            </CardHeader>
   293→   293→            <CardContent className="space-y-3">
   294→   294→              {Array.from({ length: 5 }).map((_, i) => (
   295→   295→                <Skeleton key={i} className="h-6 w-full" />
   296→   296→              ))}
   297→   297→            </CardContent>
   298→   298→          </Card>
   299→   299→          <Card>
   300→   300→            <CardHeader>
   301→   301→              <Skeleton className="h-5 w-32" />
   302→   302→              <Skeleton className="h-4 w-40" />
   303→   303→            </CardHeader>
   304→   304→            <CardContent className="space-y-3">
   305→   305→              {Array.from({ length: 4 }).map((_, i) => (
   306→   306→                <Skeleton key={i} className="h-8 w-full" />
   307→   307→              ))}
   308→   308→            </CardContent>
   309→   309→          </Card>
   310→   310→        </div>
   311→   311→      </div>
   312→   312→    );
   313→   313→  }
   314→   314→
   315→   315→  // ---- Error state ----
   316→   316→  if (error) {
   317→   317→    return (
   318→   318→      <div className="space-y-6">
   319→   319→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   320→   320→          <div>
   321→   321→            <h1 className="text-2xl font-bold tracking-tight">Dashboard</h1>
   322→   322→            <p className="text-sm text-muted-foreground">
   323→   323→              Tender activity at a glance
   324→   324→            </p>
   325→   325→          </div>
   326→   326→        </div>
   327→   327→        <Alert variant="destructive">
   328→   328→          <AlertTriangle className="h-4 w-4" />
   329→   329→          <AlertTitle>Couldn&apos;t load your dashboard</AlertTitle>
   330→   330→          <AlertDescription>
   331→   331→            <p className="mb-3">{error}</p>
   332→   332→            <Button
   333→   333→              size="sm"
   334→   334→              variant="outline"
   335→   335→              onClick={() => fetchData()}
   336→   336→              className="border-destructive/40 text-destructive hover:bg-destructive/10"
   337→   337→            >
   338→   338→              <RefreshCw className="h-3.5 w-3.5" />
   339→   339→              Try again
   340→   340→            </Button>
   341→   341→          </AlertDescription>
   342→   342→        </Alert>
   343→   343→      </div>
   344→   344→    );
   345→   345→  }
   346→   346→
   347→   347→  if (!data) return null;
   348→   348→
   349→   349→  const docsExpiringTotal = data.docs_expiring_soon + data.docs_expired;
   350→   350→
   351→   351→  return (
   352→   352→    <div className="space-y-6">
   353→   353→      {/* ---- Page header ---- */}

   355→   355→        <div>
   356→   356→          <h1 className="text-2xl font-bold tracking-tight">Dashboard</h1>
   357→   357→          <p className="text-sm text-muted-foreground">
   358→   358→            Tender activity at a glance
   359→   359→          </p>

   374→   374→            New Tender

   376→   376→        </div>
   377→   377→      </div>
   378→   378→
   379→   379→      {/* ---- KPI cards ---- */}
   380→   380→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   381→   381→        <KpiCard
   382→   382→          icon={FileStack}

   384→   384→          value={data.active_count}
   385→   385→          label="Active Tenders"
   386→   386→          subtitle="In progress"
   387→   387→        />
   388→   388→        <KpiCard
   389→   389→          icon={CalendarClock}
   390→   390→          iconClass="bg-amber-100 text-amber-600 dark:bg-amber-950 dark:text-amber-400"
   391→   391→          value={data.closing_this_week}

   398→   398→          value={docsExpiringTotal}
   399→   399→          label="Documents Expiring"
   400→   400→          subtitle="Need attention"
   401→   401→        />
   402→   402→        <KpiCard
   403→   403→          icon={Trophy}
   404→   404→          iconClass="bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400"
   405→   405→          value={data.awarded_this_year}
   406→   406→          label="Awarded This Year"
   407→   407→          subtitle="Wins"
   408→   408→        />
   409→   409→      </div>
   410→   410→
   411→   411→      {/* ---- Three-column section ---- */}
   412→   412→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   413→   413→        {/* Left column (span 2) */}
   414→   414→        <div className="space-y-4 lg:col-span-2">
   415→   415→          {/* Tender Status Breakdown */}
   416→   416→          <Card>
   417→   417→            <CardHeader>
   418→   418→              <CardTitle className="flex items-center gap-2">
   419→   419→                <FileStack className="h-4 w-4 text-primary" />
   420→   420→                Tender Status Breakdown
   421→   421→              </CardTitle>
   422→   422→              <CardDescription>
   423→   423→                Distribution across {formatNumber(data.total_tenders)} tender
   424→   424→                {data.total_tenders === 1 ? "" : "s"}
   425→   425→              </CardDescription>
   426→   426→            </CardHeader>
   427→   427→            <CardContent className="space-y-2.5">
   428→   428→              {statusRows.length === 0 ? (
   429→   429→                <EmptyState
   430→   430→                  icon={FileStack}
   431→   431→                  message="No tenders yet. Create your first tender to see a status breakdown here."
   432→   432→                />
   433→   433→              ) : (
   434→   434→                statusRows.map(([status, count]) => (
   435→   435→                  <div
   436→   436→                    key={status}
   437→   437→                    className="flex items-center gap-3 text-sm"
   438→   438→                  >
   439→   439→                    <div className="w-40 shrink-0 truncate font-medium text-foreground">
   440→   440→                      {status}
   441→   441→                    </div>
   442→   442→                    <div className="relative h-6 flex-1 overflow-hidden rounded-md bg-muted">
   443→   443→                      <div
   444→   444→                        className={cn(

   446→   446→                          statusBarColor(status)
   447→   447→                        )}
   448→   448→                        style={{
   449→   449→                          width: `${Math.max(

   452→   452→                          )}%`,
   453→   453→                        }}
   454→   454→                      />
   455→   455→                    </div>
   456→   456→                    <div className="w-10 shrink-0 text-right tabular-nums font-medium text-foreground">
   457→   457→                      {formatNumber(count)}

   460→   460→                ))

   463→   463→          </Card>
   464→   464→
   465→   465→          {/* Tenders Closing This Week */}
   466→   466→          <Card>
   467→   467→            <CardHeader>
   468→   468→              <CardTitle className="flex items-center gap-2">
   469→   469→                <CalendarClock className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   470→   470→                Tenders Closing This Week
   471→   471→              </CardTitle>
   472→   472→              <CardDescription>
   473→   473→                {data.closing_this_week > 0

   476→   476→                    } need attention`
   477→   477→                  : "All clear for this week"}
   478→   478→              </CardDescription>
   479→   479→            </CardHeader>
   480→   480→            <CardContent>
   481→   481→              {data.closing_this_week_list.length === 0 ? (
   482→   482→                <EmptyState
   483→   483→                  icon={CheckCircle2}

   486→   486→                />
