   137→   137→function dateHeading(iso: string): { label: string; relative: string | null; diff: number } {
   138→   138→  const d = new Date(iso);
   139→   139→  if (isNaN(d.getTime())) return { label: formatDate(iso), relative: null, diff: 0 };
   140→   140→  const today = new Date();
   141→   141→  today.setHours(0, 0, 0, 0);
   142→   142→  d.setHours(0, 0, 0, 0);
   143→   143→  const diff = Math.round((d.getTime() - today.getTime()) / 86400000);
   144→   144→  const label = d.toLocaleDateString("en-ZA", {
   145→   145→    weekday: "short",
   146→   146→    day: "2-digit",
   147→   147→    month: "short",
   148→   148→    year: "numeric",
   149→   149→  });
   150→   150→  let relative: string | null = null;
   151→   151→  if (diff === 0) relative = "Today";
   152→   152→  else if (diff === 1) relative = "Tomorrow";
   153→   153→  else if (diff === -1) relative = "Yesterday";
   154→   154→  else if (diff > 1 && diff <= 7) relative = `In ${diff} days`;
   155→   155→  else if (diff < 0) relative = `${Math.abs(diff)}d ago`;
   156→   156→  return { label, relative, diff };
   157→   157→}
   158→   158→
   159→   159→function relativeBadgeClass(diff: number): string {
   160→   160→  if (diff < 0)
   161→   161→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   162→   162→  if (diff <= 1)
   163→   163→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   164→   164→  if (diff <= 3)
   165→   165→    return "bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-900";
   166→   166→  return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-900";
   167→   167→}
