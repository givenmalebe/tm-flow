   133→// ---- Helpers ----
   134→function lineTotal(qty: number, rate: number): number {
   135→  const q = Number.isFinite(qty) ? qty : 0;
   136→  const r = Number.isFinite(rate) ? rate : 0;
   137→  return Math.round(q * r * 100) / 100;
   138→}
   139→
   140→function calcTotals(items: PricingLineItem[]) {
   141→  const subtotal = items.reduce(
   142→    (sum, it) => sum + lineTotal(it.quantity, it.rate),
   143→    0
   144→  );
   145→  const vat = Math.round(subtotal * VAT_RATE * 100) / 100;
   146→  const total = Math.round((subtotal + vat) * 100) / 100;
   147→  return { subtotal, vat, total };
   148→}
   149→
   150→function categorySubtotals(items: PricingLineItem[]) {
   151→  const map = new Map<string, number>();
   152→  for (const c of LINE_ITEM_CATEGORIES) map.set(c, 0);
   153→  for (const it of items) {
   154→    const key = (it.category || "Other") as string;
   155→    const prev = map.get(key) ?? 0;
   156→    map.set(key, prev + lineTotal(it.quantity, it.rate));
   157→  }
   158→  return map;
   159→}

   258→// ---- Category subtotal strip ----
   259→function CategorySubtotals({ items }: { items: PricingLineItem[] }) {
   260→  const subs = categorySubtotals(items);
   261→  const activeCategories = LINE_ITEM_CATEGORIES.filter(
   262→    (c) => (subs.get(c) ?? 0) > 0
   263→  );
   264→  if (activeCategories.length === 0) return null;
   265→  return (
   266→    <div className="flex flex-wrap items-stretch gap-2">
   267→      {activeCategories.map((c) => (
   268→        <div
   269→          key={c}
   270→          className="flex-1 min-w-[120px] rounded-lg border bg-muted/20 px-3 py-2"
   271→        >
   272→          <div className="flex items-center gap-1.5">
   273→            <CategoryBadge category={c} />
   274→          </div>
   275→          <div className="mt-1 text-sm font-semibold tabular-nums text-foreground">
   276→            {formatCurrency(subs.get(c) ?? 0)}
