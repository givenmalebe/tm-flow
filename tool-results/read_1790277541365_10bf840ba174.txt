   107→   107→function deadlineBadgeClass(days: number): string {
   108→   108→  if (days < 0 || days <= 1)
   109→   109→    return "bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-900";
   110→   110→  if (days <= 3)
   111→   111→    return "bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-900";
   112→   112→  return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-900";
   113→   113→}
   114→   114→
   115→   115→function deadlineLabel(days: number): string {
   116→   116→  if (days < 0) return `Overdue by ${Math.abs(days)}d`;
   117→   117→  if (days === 0) return "Due today";
   118→   118→  if (days === 1) return "1 day left";
   119→   119→  return `${days} days left`;
   120→   120→}
   121→   121→
   122→   122→function docExpiryLabel(days: number): string {
   123→   123→  if (days < 0) return `Expired ${Math.abs(days)}d ago`;
   124→   124→  if (days === 0) return "Expires today";
   125→   125→  if (days === 1) return "Expires tomorrow";
   126→   126→  return `In ${days}d`;
   127→   127→}

   167→   167→function KpiCard({ icon: Icon, iconClass, value, label, subtitle, onClick, hint }: KpiCardProps) {

   172→   172→      asChild={clickable}

   174→   174→        "text-left transition-all",

   176→   176→          ? "cursor-pointer hover:shadow-md hover:border-primary/40 hover:-translate-y-0.5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"

   200→   200→          >
   201→   201→            <Icon className="h-5 w-5" />
