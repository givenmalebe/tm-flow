   100→/* ------------------------------------------------------------------ */
   101→/*  localStorage helpers for "My Tenders" + progress                  */
   102→/* ------------------------------------------------------------------ */
   103→
   104→const MY_TENDERS_KEY = "tmflow_my_tenders";
   105→
   106→function getMyTenders(): Record<number, number> {
   107→  if (typeof window === "undefined") return {};
   108→  try {
   109→    const raw = localStorage.getItem(MY_TENDERS_KEY);
   110→    return raw ? JSON.parse(raw) : {};
   111→  } catch {
   112→    return {};
   113→  }
   114→}
   115→
   116→function saveMyTender(tenderId: number) {
   117→  const map = getMyTenders();
   118→  map[tenderId] = 0;
   119→  localStorage.setItem(MY_TENDERS_KEY, JSON.stringify(map));
   120→}
   121→
   122→function removeMyTender(tenderId: number) {
   123→  const map = getMyTenders();
   124→  delete map[tenderId];
   125→  localStorage.setItem(MY_TENDERS_KEY, JSON.stringify(map));
   126→}
   127→
   128→function setMyTenderProgress(tenderId: number, progress: number) {
   129→  const map = getMyTenders();
   130→  map[tenderId] = progress;
   131→  localStorage.setItem(MY_TENDERS_KEY, JSON.stringify(map));
   132→}

   168→      setLoading(false);

   173→    fetchTenders();

   184→      t.reference_number?.toLowerCase().includes(q) ||
   185→      t.company_name?.toLowerCase().includes(q) ||
   186→      t.category?.toLowerCase().includes(q) ||
   187→      t.location?.toLowerCase().includes(q)

   203→    }

   220→

   227→      setCompliance(result);
   228→    } catch (e: any) {
   229→      toast.error("Compliance check failed", { description: e.message });
   230→    } finally {
   231→      setCheckingCompliance(false);
   232→    }
   233→  };
   234→
   235→  /* ---- Download PDF ---- */
   236→  const downloadPDF = async (t: PublicTender) => {
   237→    try {
   238→      const full = await publicApi.getTender(t.id);
   239→      const { jsPDF } = await import("jspdf");
   240→      const doc = new jsPDF();
   241→      const pw = doc.internal.pageSize.getWidth();
   242→      const ph = doc.internal.pageSize.getHeight();
   243→      const m = 18;
   244→      const cw = pw - m * 2;
   245→      let y = 0;
   246→
   247→      // Header
   248→      doc.setFillColor(15, 118, 110); doc.rect(0, 0, pw, 28, "F");

   251→      doc.text("TM FLOW", m, 13);
   252→      doc.setFontSize(8); doc.text("TENDER NOTICE", m, 19);

   260→      const tLines = doc.splitTextToSize(full.title || "Untitled Tender", cw);
   261→      doc.text(tLines, m, y); y += tLines.length * 8 + 4;
   262→      doc.setFontSize(10); doc.setFont("helvetica", "normal"); doc.setTextColor(115, 115, 115);
   263→      doc.text(`Issued by: ${full.company_name || "—"}`, m, y); y += 10;
   264→
   265→      // Sections
   266→      const sections: [string, string][] = [
   267→        ["DESCRIPTION", full.description || ""],
   268→        ["SCOPE OF WORK", full.scope_of_work || ""],
   269→        ["EVALUATION CRITERIA", full.evaluation_criteria || ""],
   270→        ["SUBMISSION METHOD", full.submission_method || ""],
   271→        ["FULL TENDER DOCUMENT", full.ai_generated_content || ""],
   272→      ];
   273→      for (const [label, text] of sections) {
   274→        if (!text) continue;
   275→        if (y > ph - 40) { doc.addPage(); y = m; }
   276→        doc.setFillColor(217, 119, 6); doc.roundedRect(m, y, 3, 5, 1, 1, "F");
   277→        doc.setFontSize(9); doc.setFont("helvetica", "bold"); doc.setTextColor(24, 24, 27);
   278→        doc.text(label, m + 6, y + 4);
   279→        doc.setDrawColor(228, 228, 231); doc.line(m, y + 7, pw - m, y + 7);
   280→        y += 12;
   281→        doc.setFontSize(8); doc.setFont("helvetica", "normal"); doc.setTextColor(51, 51, 51);
   282→        const lines = doc.splitTextToSize(text, cw);
   283→        lines.forEach((line: string) => {
   284→          if (y > ph - 20) { doc.addPage(); y = m; }
   285→          doc.text(line, m, y); y += 4.5;
   286→        });
   287→        y += 5;
   288→      }
   289→
   290→      // Footer
   291→      const pc = doc.getNumberOfPages();
   292→      for (let i = 1; i <= pc; i++) {
   293→        doc.setPage(i);
   294→        doc.setDrawColor(228, 228, 231); doc.line(m, ph - 14, pw - m, ph - 14);
   295→        doc.setFontSize(6); doc.setTextColor(150, 150, 150);
   296→        doc.text(`TM Flow | ${full.reference_number || ""} | Page ${i} of ${pc}`, m, ph - 8);
   297→      }
   298→      doc.save(`${full.reference_number || "tender"}.pdf`);
   299→      toast.success("PDF downloaded");
   300→    } catch (e: any) {

   302→    }

   304→
   305→  /* ---- Stats ---- */

   307→  const addedCount = Object.keys(myTenders).length;

   311→  /*  Render                                                            */

   313→
   314→  return (

   316→      {/* Header */}
   317→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   318→        <div>
   319→          <h1 className="text-2xl font-bold flex items-center gap-2">
   320→            <Radar className="h-6 w-6 text-primary" />
   321→            Discover Tenders
   322→          </h1>

   324→            Browse live tender opportunities published on the TM Flow platform.
   325→          </p>
   326→        </div>
   327→      </div>
   328→
   329→      {/* Stats */}
   330→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">

   332→          <CardContent className="p-4 flex items-center gap-3">

   335→            </div>
   336→            <div>
   337→              <div className="text-2xl font-bold">{totalTenders}</div>
   338→              <div className="text-xs text-muted-foreground">Tenders Available</div>
   339→            </div>
   340→          </CardContent>
   341→        </Card>

   343→          <CardContent className="p-4 flex items-center gap-3">
   344→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   345→              <Building2 className="h-5 w-5 text-amber-600" />
   346→            </div>
   347→            <div>

   349→              <div className="text-xs text-muted-foreground">Issuing Organisations</div>
   350→            </div>
   351→          </CardContent>
   352→        </Card>
   353→        <Card>
   354→          <CardContent className="p-4 flex items-center gap-3">

   356→              <CheckCircle2 className="h-5 w-5 text-emerald-600" />
   357→            </div>
   358→            <div>
   359→              <div className="text-2xl font-bold">{addedCount}</div>
   360→              <div className="text-xs text-muted-foreground">Added to My Tenders</div>
   361→            </div>
   362→          </CardContent>
   363→        </Card>
   364→      </div>
   365→
   366→      {/* Search + filter */}
   367→      <Card>
   368→        <CardContent className="p-4 space-y-3">
   369→          <div className="flex flex-col sm:flex-row gap-3">
   370→            <div className="relative flex-1">
   371→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   372→              <Input
   373→                placeholder="Search by title, reference, company, category, location…"
   374→                value={search}
   375→                onChange={(e) => setSearch(e.target.value)}
   376→                className="pl-9"

   378→            </div>
   379→            <Select value={category} onValueChange={setCategory}>
   380→              <SelectTrigger className="w-full sm:w-56">
   381→                <Filter className="h-4 w-4 mr-2 text-muted-foreground" />
   382→                <SelectValue placeholder="All categories" />
   383→              </SelectTrigger>
   384→              <SelectContent>
   385→                <SelectItem value="All">All categories</SelectItem>
   386→                {categories.map((c) => (
   387→                  <SelectItem key={c} value={c}>{c}</SelectItem>
   388→                ))}
   389→              </SelectContent>
   390→            </Select>
   391→          </div>
   392→        </CardContent>
   393→      </Card>
   394→
   395→      {/* Error */}
   396→      {error && (
   397→        <Alert variant="destructive">
   398→          <AlertTriangle className="h-4 w-4" />
   399→          <AlertDescription>{error}</AlertDescription>
   400→        </Alert>
   401→      )}
   402→
   403→      {/* Tenders grid */}
   404→      {loading ? (
   405→        <div className="flex flex-col items-center justify-center py-16">

   408→        </div>

   411→          <CardContent className="p-12 text-center">
   412→            <FileText className="h-16 w-16 text-muted-foreground/30 mx-auto mb-4" />
   413→            <h3 className="text-lg font-semibold mb-1">
   414→              {totalTenders === 0 ? "No tenders published yet" : "No results found"}
   415→            </h3>
   416→            <p className="text-sm text-muted-foreground max-w-md mx-auto">
   417→              {totalTenders === 0
   418→                ? "Check back soon for new opportunities from issuing organisations."
   419→                : "Try a different search term or category filter."}
   420→            </p>
   421→            {search && (
   422→              <Button variant="outline" className="mt-4" onClick={() => { setSearch(""); setCategory("All"); }}>
   423→                Clear filters
   424→              </Button>
   425→            )}
   426→          </CardContent>
   427→        </Card>
   428→      ) : (
   429→        <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   430→          {filtered.map((t) => {
   431→            const days = daysUntil(t.closing_date);
   432→            const isAdded = myTenders[t.id] !== undefined;
   433→            const progress = myTenders[t.id] ?? 0;

   508→                        </span>

   516→                  <div className="flex gap-2 mt-4">
   517→                    <Button
   518→                      size="sm"
   519→                      variant="outline"
   520→                      className="flex-1 text-xs"
   521→                      onClick={() => openDetail(t)}

   798→function SubmitBidDialog({
   799→  tender,
   800→  open,
   801→  onOpenChange,
   802→  onSubmitted,
   803→}: {
   804→  tender: PublicTender | null;
   805→  open: boolean;
   806→  onOpenChange: (o: boolean) => void;
   807→  onSubmitted: () => void;
   808→}) {
   809→  const [companyName, setCompanyName] = useState("");
   810→  const [contactPerson, setContactPerson] = useState("");
   811→  const [email, setEmail] = useState("");
   812→  const [phone, setPhone] = useState("");
   813→  const [bbbeeLevel, setBbbeeLevel] = useState("");
   814→  const [amount, setAmount] = useState("");
   815→  const [summary, setSummary] = useState("");
   816→  const [files, setFiles] = useState<File[]>([]);
   817→  const [submitting, setSubmitting] = useState(false);
   818→
   819→  useEffect(() => {
   820→    if (open && tender) {
   821→      // Pre-fill from logged-in user if possible
   822→      try {
   823→        const raw = localStorage.getItem("tenderflow_user");
   824→        if (raw) {
   825→          const u = JSON.parse(raw);
   826→          setContactPerson(u.full_name || "");
   827→          setCompanyName(u.company_name || "");

   834→  const handleSubmit = async (e: React.FormEvent) => {
   835→    e.preventDefault();
   836→    if (!tender) return;
   837→    if (!companyName.trim() || !contactPerson.trim()) {
   838→      toast.error("Company name and contact person are required");

   857→    } catch (err: any) {
   858→      toast.error("Failed to submit bid", { description: err.message });
   859→    } finally {
   860→      setSubmitting(false);
   861→    }
   862→  };
   863→
   864→  return (
   865→    <Dialog open={open} onOpenChange={onOpenChange}>
