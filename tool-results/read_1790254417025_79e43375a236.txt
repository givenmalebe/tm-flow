   120→// ---- Helpers ----
   121→function lineTotal(qty: number, rate: number): number {
   122→  const q = Number.isFinite(qty) ? qty : 0;
   123→  const r = Number.isFinite(rate) ? rate : 0;
   124→  return Math.round(q * r * 100) / 100;
   125→}
   126→
   127→function calcTotals(items: PricingLineItem[]) {
   128→  const subtotal = items.reduce(
   129→    (sum, it) => sum + lineTotal(it.quantity, it.rate),
   130→    0
   131→  );
   132→  const vat = Math.round(subtotal * VAT_RATE * 100) / 100;
   133→  const total = Math.round((subtotal + vat) * 100) / 100;
   134→  return { subtotal, vat, total };
   135→}

   705→              onChange={(e) => setNotes(e.target.value)}
   706→              placeholder="Assumptions, exclusions, terms…"
   707→              rows={3}
   708→            />
   709→          </div>
   710→        </div>
   711→
   712→        <DialogFooter className="gap-2">
   713→          <Button
   714→            variant="outline"
   715→            onClick={() => onOpenChange(false)}
   716→            disabled={saving}
   717→          >

   719→          </Button>
   720→          <Button onClick={handleSave} disabled={saving || !hasAnyItem}>
   721→            {saving ? (
   722→              <>
   723→                <Loader2 className="h-4 w-4 animate-spin" />
   724→                Saving…
   725→              </>
   726→            ) : (

   728→                <Save className="h-4 w-4" />

   731→            )}
   732→          </Button>

   739→// ---- Main Component ----
   740→export function PricingTab() {
   741→  const [tenders, setTenders] = useState<TenderOption[]>([]);
   742→  const [selectedTenderId, setSelectedTenderId] = useState<number | null>(null);
   743→  const [versions, setVersions] = useState<PricingVersion[]>([]);
   744→  const [loadingTenders, setLoadingTenders] = useState(true);
   745→  const [loadingVersions, setLoadingVersions] = useState(false);
   746→  const [refreshing, setRefreshing] = useState(false);
   747→  const [error, setError] = useState<string | null>(null);
   748→  const [dialogOpen, setDialogOpen] = useState(false);
   749→
   750→  const fetchTenders = useCallback(async (isRefresh = false) => {
   751→    if (isRefresh) setRefreshing(true);
   752→    else setLoadingTenders(true);
   753→    setError(null);
   754→    try {
   755→      const res = await api.listTenders({ sort: "updated_at", order: "desc" });
   756→      setTenders(
   757→        res.items.map((t) => ({
   758→          id: t.id,
   759→          reference_number: t.reference_number,
   760→          title: t.title,
   761→          client_company: t.client_company,
   762→          status: t.status,
   763→        }))
   764→      );
   765→    } catch (e: any) {
   766→      setError(e?.message || "Failed to load tenders");
   767→    } finally {
   768→      setLoadingTenders(false);
   769→      setRefreshing(false);
   770→    }
   771→  }, []);
   772→
   773→  const fetchVersions = useCallback(async (tenderId: number) => {
   774→    setLoadingVersions(true);
   775→    try {
   776→      const res = await api.listPricing(tenderId);
   777→      const list = (res.items || []) as PricingVersion[];
   778→      list.sort((a, b) => b.version_number - a.version_number);
   779→      setVersions(list);
   780→    } catch (e: any) {
   781→      toast.error(e?.message || "Failed to load pricing versions");
   782→      setVersions([]);
   783→    } finally {
   784→      setLoadingVersions(false);
   785→    }
   786→  }, []);
   787→
   788→  useEffect(() => {
   789→    fetchTenders();
   790→  }, [fetchTenders]);
   791→
   792→  useEffect(() => {
   793→    if (selectedTenderId !== null) {
   794→      fetchVersions(selectedTenderId);
   795→    } else {
   796→      setVersions([]);
   797→    }
   798→  }, [selectedTenderId, fetchVersions]);
   799→
   800→  const selectedTender = useMemo(
   801→    () => tenders.find((t) => t.id === selectedTenderId) || null,
   802→    [tenders, selectedTenderId]
   803→  );
   804→
   805→  const nextVersionNumber = useMemo(() => {
   806→    if (versions.length === 0) return 1;
   807→    return Math.max(...versions.map((v) => v.version_number)) + 1;
   808→  }, [versions]);
   809→
   810→  const totalValue = useMemo(
   811→    () =>
   812→      versions.reduce((sum, v) => sum + (v.total || 0), 0),
   813→    [versions]
   814→  );
   815→  const latestTotal = useMemo(
   816→    () => (versions.length > 0 ? versions[0].total : 0),
   817→    [versions]
   818→  );
   819→
   820→  const handleDelete = async (versionId: number) => {
   821→    if (selectedTenderId === null) return;
   822→    try {
   823→      await api.deletePricing(versionId);
   824→      toast.success("Pricing version deleted");
   825→      fetchVersions(selectedTenderId);
   826→    } catch (e: any) {
   827→      toast.error(e?.message || "Failed to delete pricing version");
   828→    }
   829→  };
   830→
   831→  const handleRefresh = () => {
   832→    fetchTenders(true);
   833→    if (selectedTenderId !== null) {
   834→      fetchVersions(selectedTenderId);
   835→    }
   836→  };
   837→
   838→  // ---- Loading state ----
   839→  if (loadingTenders) {
   840→    return (
   841→      <div className="space-y-6">
   842→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   843→          <div className="space-y-2">
   844→            <Skeleton className="h-8 w-48" />
   845→            <Skeleton className="h-4 w-64" />
   846→          </div>
   847→          <Skeleton className="h-9 w-32" />
   848→        </div>
   849→        <Skeleton className="h-10 w-full max-w-md" />
   850→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   851→          {Array.from({ length: 3 }).map((_, i) => (
   852→            <Card key={i}>
   853→              <CardContent className="flex items-start justify-between gap-3">
   854→                <div className="flex-1 space-y-2">
   855→                  <Skeleton className="h-7 w-20" />
   856→                  <Skeleton className="h-4 w-24" />
   857→                </div>
   858→                <Skeleton className="h-11 w-11 rounded-xl" />
   859→              </CardContent>
   860→            </Card>
   861→          ))}
   862→        </div>
   863→        <Card>
   864→          <CardHeader>
   865→            <Skeleton className="h-5 w-40" />
   866→            <Skeleton className="h-4 w-56" />
   867→          </CardHeader>
   868→          <CardContent className="space-y-3">
   869→            {Array.from({ length: 3 }).map((_, i) => (
   870→              <Skeleton key={i} className="h-24 w-full" />
   871→            ))}
   872→          </CardContent>
   873→        </Card>
   874→      </div>
   875→    );
   876→  }
   877→
   878→  // ---- Error state ----
   879→  if (error) {
   880→    return (
   881→      <div className="space-y-6">
   882→        <div>
   883→          <h1 className="text-2xl font-bold tracking-tight">Pricing Tracking</h1>
   884→          <p className="text-sm text-muted-foreground">
   885→            Build pricing sheets and track revisions
   886→          </p>
   887→        </div>
   888→        <Alert variant="destructive">
   889→          <AlertTriangle className="h-4 w-4" />
   890→          <AlertTitle>Couldn&apos;t load pricing data</AlertTitle>
   891→          <AlertDescription>
   892→            <p className="mb-3">{error}</p>
   893→            <Button
   894→              size="sm"
   895→              variant="outline"
   896→              onClick={() => fetchTenders()}
   897→              className="border-destructive/40 text-destructive hover:bg-destructive/10"
   898→            >
   899→              <RefreshCw className="h-3.5 w-3.5" />
   900→              Try again
   901→            </Button>
   902→          </AlertDescription>
   903→        </Alert>
   904→      </div>
   905→    );
   906→  }
   907→
   908→  return (
   909→    <div className="space-y-6">
   910→      {/* ---- Page header ---- */}
   911→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   912→        <div>
   913→          <h1 className="text-2xl font-bold tracking-tight">Pricing Tracking</h1>
   914→          <p className="text-sm text-muted-foreground">
   915→            Build pricing sheets and track revisions
   916→          </p>
   917→        </div>
   918→        <Button
   919→          variant="ghost"
   920→          onClick={handleRefresh}
   921→          disabled={refreshing}
   922→        >
   923→          <RefreshCw className={cn("h-4 w-4", refreshing && "animate-spin")} />
   924→          Refresh
   925→        </Button>
   926→      </div>
   927→
   928→      {/* ---- Tender selector ---- */}
   929→      <div className="grid gap-2">
   930→        <Label className="flex items-center gap-1.5 text-sm font-medium">
   931→          <Hash className="h-3.5 w-3.5 text-primary" />
   932→          Select tender
   933→        </Label>
   934→        {tenders.length === 0 ? (
   935→          <Alert>
   936→            <AlertTriangle className="h-4 w-4" />
   937→            <AlertTitle>No tenders available</AlertTitle>

   942→        ) : (
   943→          <Select
   944→            value={selectedTenderId !== null ? String(selectedTenderId) : undefined}
   945→            onValueChange={(v) => setSelectedTenderId(v ? Number(v) : null)}
   946→          >
   947→            <SelectTrigger className="w-full max-w-xl">

   950→            <SelectContent>
   951→              {tenders.map((t) => (
   952→                <SelectItem key={t.id} value={String(t.id)}>
   953→                  <span className="font-mono text-xs font-semibold text-primary mr-2">
   954→                    {t.reference_number}
   955→                  </span>

   983→                    <div className="flex flex-wrap items-center gap-2">

   985→                        {selectedTender.reference_number}

   987→                      <Badge
   988→                        className={cn(
   989→                          "text-[10px] font-medium",
   990→                          statusColor(selectedTender.status)
   991→                        )}
   992→                      >
   993→                        {selectedTender.status}

   996→                    <div className="text-base font-semibold text-foreground line-clamp-1">

  1057→                />
  1058→              </CardContent>
  1059→            </Card>

  1061→            <div className="space-y-3">

  1063→                <VersionCard

  1066→                  onDelete={handleDelete}
  1067→                  defaultOpen={idx === 0}

  1075→      {/* ---- New Pricing Version Dialog ---- */}
  1076→      {selectedTenderId !== null && (
  1077→        <NewPricingDialog
  1078→          open={dialogOpen}
  1079→          onOpenChange={setDialogOpen}
  1080→          tenderId={selectedTenderId}

  1082→          onSaved={() => fetchVersions(selectedTenderId)}
  1083→        />
  1084→      )}
  1085→    </div>
  1086→  );
  1087→}
